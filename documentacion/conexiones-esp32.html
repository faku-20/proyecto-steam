Conexiones y codigo del prototipo ESP32

Sistema de riego para curado de ladrillos ecologicos. Documento para poder desarmar y volver a armar el prototipo probado.

ESP32 DEVKIT V1LCD 16x2Sensor humedadRele GPIO 26LEDs GPIO 32/33Botones GPIO 25/27Buzzer por rele

Estado actual: queda documentado el control por web local, sensor de humedad, LCD 16x2, rele en GPIO 26, LED rojo de apagado en GPIO 32, LED verde de encendido en GPIO 33, boton de encendido en GPIO 25, boton de apagado en GPIO 27 y buzzer conectado a la salida del rele.

6. Rele, LEDs, buzzer y botones

Rele

Modulo releConectar a
S / INGPIO 26
+ / VCCVN / 5V
- / GNDGND

El codigo esta configurado para rele activo en LOW. Si funciona al reves, cambiar RELAY_ACTIVE_LOW a false.

LED rojo y LED verde

ElementoConexion
LED rojo OFFGPIO 32 -> resistencia 330 ohm -> pata larga LED rojo; pata corta -> GND
LED verde ONGPIO 33 -> resistencia 330 ohm -> pata larga LED verde; pata corta -> GND

Buzzer por contactos del rele

ConexionDestino
VN / 5VCOM del rele
NO del relePositivo del buzzer
Negativo del buzzerGND

Usar NO, no NC. Asi el buzzer queda apagado por defecto y suena solo cuando el rele se activa.

Botones fisicos

BotonConexionFuncion
Encender / regarUna pata a GPIO 25, otra a GNDInicia riego de 5 segundos
ApagarUna pata a GPIO 27, otra a GNDApagado inmediato

No llevan resistencia externa porque el codigo usa INPUT_PULLUP.

7. Seguridad

Seguridad: no conectar 220V en protoboard ni directamente al ESP32. Para bomba real/contacto de potencia, hacerlo solo con supervision profesional, gabinete, borneras y protecciones.

8. Esquema general

ESP32
 |-- GPIO 34  <- Sensor humedad AOUT
 |-- GPIO 26  -> Rele S/IN
 |-- GPIO 25  <- Boton encender a GND
 |-- GPIO 27  <- Boton apagar a GND
 |-- GPIO 32  -> LED rojo OFF con resistencia
 |-- GPIO 33  -> LED verde ON con resistencia
 |-- GPIO 19  -> LCD RS
 |-- GPIO 23  -> LCD E
 |-- GPIO 18  -> LCD D4
 |-- GPIO 21  -> LCD D5
 |-- GPIO 22  -> LCD D6
 |-- GPIO 13  -> LCD D7
 |-- 3V3      -> Sensor VCC
 |-- VN/5V    -> LCD VDD, LCD A, rele +, potenciometro, COM rele para buzzer
 |-- GND      -> LCD VSS/RW/K, sensor GND, rele -, botones, LEDs y buzzer

9. Codigo completo actualizado

#include <WiFi.h>
#include <WebServer.h>
#include <LiquidCrystal.h>

const char* ssid = "CuradoLadrillos";
const char* password = "12345678";

WebServer server(80);

// LCD: RS, E, D4, D5, D6, D7
LiquidCrystal lcd(19, 23, 18, 21, 22, 13);

const int HUMIDITY_PIN = 34;
const int RELAY_PIN = 26;
const int BUTTON_ON_PIN = 25;
const int BUTTON_OFF_PIN = 27;
const int RED_LED_PIN = 32;
const int GREEN_LED_PIN = 33;

// Si el rele funciona al reves, cambiar a false.
const bool RELAY_ACTIVE_LOW = true;

const int VALOR_SECO = 2463;
const int VALOR_HUMEDO = 959;

bool pumpOn = false;
unsigned long pumpStartTime = 0;
unsigned long pumpDuration = 0;
unsigned long lastLcdUpdate = 0;
unsigned long lastButtonRead = 0;

int readHumidityPercent() {
  int raw = analogRead(HUMIDITY_PIN);
  int humedad = map(raw, VALOR_SECO, VALOR_HUMEDO, 0, 100);
  return constrain(humedad, 0, 100);
}

void writeRelay(bool on) {
  if (RELAY_ACTIVE_LOW) {
    digitalWrite(RELAY_PIN, on ? LOW : HIGH);
  } else {
    digitalWrite(RELAY_PIN, on ? HIGH : LOW);
  }
}

void setPump(bool on) {
  pumpOn = on;
  writeRelay(on);

  digitalWrite(RED_LED_PIN, on ? LOW : HIGH);
  digitalWrite(GREEN_LED_PIN, on ? HIGH : LOW);

  Serial.println(on ? "Rele ACTIVADO" : "Rele APAGADO");
}

unsigned long getRemainingSeconds() {
  if (!pumpOn || pumpDuration == 0) return 0;

  unsigned long elapsed = millis() - pumpStartTime;
  if (elapsed >= pumpDuration) return 0;

  return (pumpDuration - elapsed + 999) / 1000;
}

void updateLcd() {
  int humedad = readHumidityPercent();
  unsigned long restante = getRemainingSeconds();

  lcd.setCursor(0, 0);
  lcd.print("Hum: ");
  lcd.print(humedad);
  lcd.print("%       ");

  lcd.setCursor(0, 1);
  if (pumpOn) {
    lcd.print("Riego: ");
    lcd.print(restante);
    lcd.print("s       ");
  } else {
    lcd.print("Bomba: OFF      ");
  }
}

void startIrrigation(unsigned long durationMs) {
  pumpStartTime = millis();
  pumpDuration = durationMs;
  setPump(true);

  Serial.print("Riego iniciado por ");
  Serial.print(durationMs / 1000);
  Serial.println(" segundos");
}

void stopIrrigation() {
  pumpDuration = 0;
  setPump(false);
  Serial.println("Riego detenido");
}

String htmlPage() {
  int humedad = readHumidityPercent();
  String estado = pumpOn ? "ENCENDIDA" : "APAGADA";
  unsigned long restante = getRemainingSeconds();

  String page = "";
  page += "<!DOCTYPE html><html><head>";
  page += "<meta name='viewport' content='width=device-width, initial-scale=1'>";
  page += "<meta http-equiv='refresh' content='1'>";
  page += "<title>Curado de Ladrillos</title>";
  page += "<style>";
  page += "body{font-family:Arial;background:#f2f2f2;margin:20px;color:#222;}";
  page += ".box{max-width:420px;margin:auto;background:white;padding:20px;border-radius:8px;}";
  page += "h1{font-size:22px;margin-top:0;}";
  page += ".dato{font-size:20px;margin:12px 0;}";
  page += "button{width:100%;padding:14px;margin:8px 0;font-size:18px;border:0;border-radius:6px;}";
  page += ".on{background:#1f8f4d;color:white;}";
  page += ".off{background:#b3261e;color:white;}";
  page += "</style></head><body>";
  page += "<div class='box'>";
  page += "<h1>Riego para curado</h1>";
  page += "<div class='dato'>Humedad: <b>" + String(humedad) + "%</b></div>";
  page += "<div class='dato'>Bomba: <b>" + estado + "</b></div>";

  if (pumpOn) {
    page += "<div class='dato'>Tiempo restante: <b>";
    page += String(restante);
    page += " s</b></div>";
  }

  page += "<form action='/regar5'><button class='on'>Regar 5 segundos</button></form>";

  page += "<form action='/apagar'><button class='off'>Apagar ahora</button></form>";

  page += "</div></body></html>";

}

void handleRoot() {
  server.send(200, "text/html", htmlPage());
}

void handleWater5() {
  startIrrigation(5000);
  server.sendHeader("Location", "/");
  server.send(303);
}

void handleWater10() {
  startIrrigation(10000);
  server.sendHeader("Location", "/");
  server.send(303);
}

void handleWater20() {
  startIrrigation(20000);
  server.sendHeader("Location", "/");
  server.send(303);
}

void handleStop() {
  stopIrrigation();
  server.sendHeader("Location", "/");
  server.send(303);
}

void setup() {
  Serial.begin(115200);
  delay(1000);

  pinMode(RELAY_PIN, OUTPUT);
  pinMode(BUTTON_ON_PIN, INPUT_PULLUP);
  pinMode(BUTTON_OFF_PIN, INPUT_PULLUP);
  pinMode(RED_LED_PIN, OUTPUT);
  pinMode(GREEN_LED_PIN, OUTPUT);

  setPump(false);

  lcd.begin(16, 2);
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Curado ladrillo");
  lcd.setCursor(0, 1);
  lcd.print("Iniciando...");

  Serial.println("Creando red WiFi...");
  WiFi.softAP(ssid, password);

  Serial.print("Red: ");
  Serial.println(ssid);
  Serial.print("IP: ");
  Serial.println(WiFi.softAPIP());

  server.on("/", handleRoot);
  server.on("/regar5", handleWater5);
  server.on("/regar10", handleWater10);
  server.on("/regar20", handleWater20);
  server.on("/apagar", handleStop);

  server.begin();

  delay(1000);
  updateLcd();
}

void loop() {
  server.handleClient();

  if (pumpOn && pumpDuration > 0) {
    unsigned long elapsed = millis() - pumpStartTime;

    if (elapsed >= pumpDuration) {
      stopIrrigation();
    }
  }

  if (millis() - lastButtonRead > 250) {
    if (digitalRead(BUTTON_ON_PIN) == LOW) {
      lastButtonRead = millis();
      if (!pumpOn) {
        startIrrigation(5000);
        Serial.println("Riego iniciado por boton fisico");
      }
    }

    if (digitalRead(BUTTON_OFF_PIN) == LOW) {
      lastButtonRead = millis();
      if (pumpOn) {
        stopIrrigation();
        Serial.println("Apagado por boton fisico");
      }
    }
  }

  if (millis() - lastLcdUpdate >= 1000) {
    lastLcdUpdate = millis();
    updateLcd();
  }
}

10. Orden para volver a armar

  1. Conectar alimentacion de la LCD, contraste y RW a GND.
  2. Conectar los 6 cables de datos/control de la LCD.
  3. Conectar sensor de humedad: VCC a 3V3, GND a GND, AOUT a GPIO 34.
  4. Conectar rele: S a GPIO 26, + a VN/5V, - a GND.
  5. Conectar LED rojo en GPIO 32 con resistencia.
  6. Conectar LED verde en GPIO 33 con resistencia.
  7. Conectar boton ON entre GPIO 25 y GND.
  8. Conectar boton OFF entre GPIO 27 y GND.
  9. Conectar buzzer: VN/5V a COM, NO a positivo del buzzer, negativo del buzzer a GND.
  10. Cargar el codigo completo y probar web, LCD, LEDs, botones y rele.
